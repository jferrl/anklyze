import { useMemo } from 'react';

**2. Wrap `stats` computation in `useMemo`:**

Replace the inline `stats` object (lines 32-45) with:
```tsx
const stats = useMemo(() => ({
  totalCases: cases.length,
  draftCases: cases.filter((c) => c.status === 'draft').length,
  publishedCases: cases.filter((c) => c.status === 'published').length,
  closedCases: cases.filter((c) => c.status === 'closed').length,
  totalResponses: cases.reduce((sum, c) => sum + c.response_count, 0),
  totalUniqueUsers: cases.reduce((sum, c) => sum + c.unique_users, 0),
  avgResponsesPerCase:
    cases.length > 0
      ? Math.round(
          cases.reduce((sum, c) => sum + c.response_count, 0) / cases.length
        )
      : 0,
}), [cases]);
```

**3. Wrap `recentActiveCases` in `useMemo`:**

Replace lines 47-50 with:
```tsx
const recentActiveCases = useMemo(() =>
  [...cases]
    .filter((c) => c.response_count > 0)
    .sort((a, b) => new Date(b.updated_at).getTime() - new Date(a.updated_at).getTime())
    .slice(0, 5),
  [cases]
);
```

**4. Wrap `casesNeedingAttention` in `useMemo`:**

Replace lines 52-57 with:
```tsx
const casesNeedingAttention = useMemo(() =>
  cases.filter((c) => {
    if (c.status === 'published' && c.response_count === 0) return true;
    if (c.deadline && new Date(c.deadline) < new Date() && c.status === 'published')
      return true;
    return false;
  }),
  [cases]
);
```

**Why this matters:** `AdminDashboardPage` passes these derived values as props to `RecentActivityCard`, `NeedsAttentionCard`, and `StatusBreakdown`. Without `useMemo`, every render recomputes 4 filters, 2 reduces, a sort, and a slice on the cases array. With 100 cases (the page fetches up to 100), this is wasteful on re-renders triggered by any state change.

**Why NOT memo the child components:** The children (`RecentActivityCard`, `NeedsAttentionCard`, `StatusBreakdown`) are not wrapped in `React.memo`. Wrapping them would be over-engineering for display-only components. The `useMemo` on the parent prevents the expensive array operations — the children will still re-render but with the same props when the source data hasn't changed.

```tsx
// Move OUTSIDE the component (before export function CaseAnalyticsPage)
function formatDuration(ms: number) {
  const seconds = Math.floor(ms / 1000);
  if (seconds < 60) return `${seconds}s`;
  const minutes = Math.floor(seconds / 60);
  const remainingSeconds = seconds % 60;
  return `${minutes}m ${remainingSeconds}s`;
}
```
