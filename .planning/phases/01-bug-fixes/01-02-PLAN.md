**Step 2 — Add onError to AdminCasesPage.tsx mutations:**

In `frontend/src/pages/admin/AdminCasesPage.tsx`:

Import `toast` from `sonner` and `useTranslation` from `react-i18next` (check if already imported — only add if missing).

Add `onError` callbacks to `publishMutation` and `closeMutation`:

```tsx
const publishMutation = useMutation({
  mutationFn: caseApi.publishCase,
  onSuccess: () => {
    queryClient.invalidateQueries({ queryKey: ['admin-cases'], refetchType: 'all' });
    queryClient.invalidateQueries({ queryKey: ['admin-cases-all'], refetchType: 'all' });
    queryClient.invalidateQueries({ queryKey: ['published-cases'], refetchType: 'all' });
  },
  onError: (error: Error) => {
    const apiError = error as Error & { code?: string; error_code?: string };
    const errorCode = apiError.code || apiError.error_code;
    if (errorCode === 'INVALID_STATE_TRANSITION') {
      toast.error(t('errors.invalidStateTransition'));
    } else {
      toast.error(t('errors.publishFailed'));
    }
  },
});

const closeMutation = useMutation({
  mutationFn: caseApi.closeCase,
  onSuccess: () => {
    queryClient.invalidateQueries({ queryKey: ['admin-cases'], refetchType: 'all' });
    queryClient.invalidateQueries({ queryKey: ['admin-cases-all'], refetchType: 'all' });
    queryClient.invalidateQueries({ queryKey: ['published-cases'], refetchType: 'all' });
  },
  onError: (error: Error) => {
    const apiError = error as Error & { code?: string; error_code?: string };
    const errorCode = apiError.code || apiError.error_code;
    if (errorCode === 'INVALID_STATE_TRANSITION') {
      toast.error(t('errors.invalidStateTransition'));
    } else {
      toast.error(t('errors.closeFailed'));
    }
  },
});
```

Check how other mutations in this file or other admin pages extract the error code from API responses. If the API client already puts the code on the error object differently (e.g., `error.code` directly), follow that pattern instead.
